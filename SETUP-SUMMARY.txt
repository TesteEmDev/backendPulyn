├─ DB_DRIVER=postgres

├─ JWT_SECRET=seu-segredo-local

├─ NODE_ENV=production
├─ JWT_SECRET=novo-segredo-complexo
├─ DATABASE_URL=postgresql://...
└─ FRONTEND_URL=seu-dominio.com

Gerar JWT_SECRET:
  node -e "console.log(require('crypto').randomBytes(32).toString('hex'))"
